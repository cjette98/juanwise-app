# JuanWise UI Revamp Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the per-screen ad-hoc styling of the 14 student-journey screens with a shared design-token and primitive-component layer rendering the approved Playful Heritage direction.

**Architecture:** A three-layer stack. `src/shared/theme/tokens.ts` holds every colour, spacing, radius, type and elevation value as plain data (unit-testable, no React). `src/shared/components/ui/` holds presentational primitives that consume only tokens and own no domain logic. Screens keep their existing hooks, contexts and navigation calls untouched and swap their markup for primitives. Nothing below the presentation layer changes.

**Tech Stack:** Expo 57 / React Native 0.86, expo-router, react-native-svg 15.15.4 (already present, used for icons and the sun-ray motif), expo-font 57 with `@expo-google-fonts/baloo-2` and `@expo-google-fonts/nunito` (new), vitest 4 for pure-module tests.

**Spec:** `docs/superpowers/specs/2026-09-06-ui-revamp-design.md`

## Global Constraints

- Minimum font size anywhere: **11.5**. Minimum body copy: **15**.
- Minimum tap target on its shorter axis: **46**. Primary actions: **56**.
- No emoji used as an icon. Emoji as *content* (student avatar) is allowed.
- No screen may hardcode a colour, radius, font size, font family or shadow. All come from `tokens`.
- No navigation destination is added or removed. No API call, context or game rule changes.
- The quiz mini-lesson phase shows **text only** — never the mini-lesson picture.
- Existing behaviour of `welcome-screen.tsx`'s entrance animation is preserved.
- Every task ends with `npx tsc --noEmit` clean and `npx vitest run` green before its commit.

```tsx
// paints a frame in the system font and then reflows into Baloo/Nunito.
void SplashScreen.preventAutoHideAsync();

export default function RootLayout() {
  const [fontsLoaded, fontError] = useFonts({
    Baloo2_700Bold,
    Baloo2_800ExtraBold,
    Nunito_600SemiBold,
    Nunito_700Bold,
    Nunito_800ExtraBold,
  });

  useEffect(() => {
    // A font that fails to load must not leave the splash up forever — the app
    // is perfectly usable in the fallback face.
    if (fontsLoaded || fontError) void SplashScreen.hideAsync();
  }, [fontsLoaded, fontError]);

  if (!fontsLoaded && !fontError) return null;

  return (
    // ... existing tree unchanged ...
  );
}
```

```tsx
// src/shared/components/ui/icon.tsx
import React from 'react';
import Svg, { Path } from 'react-native-svg';
import { ICONS, type IconName } from './icon-registry';
import tokens from '@/shared/theme/tokens';

export type { IconName };

/**
 * A stroked line icon. `filled` swaps stroke for fill, which is what a star or
 * a medal wants once it has been earned.
 */
export function Icon({
  name,
  size = 24,
  color = tokens.color.ink,
  strokeWidth = 2.2,
  filled = false,
}: {
  name: IconName;
  size?: number;
  color?: string;
  strokeWidth?: number;
  filled?: boolean;
}) {
  return (
    <Svg width={size} height={size} viewBox="0 0 24 24" fill="none">
      {ICONS[name].map((d, i) => (
        <Path
          key={i}
          d={d}
          stroke={filled ? 'none' : color}
          fill={filled ? color : 'none'}
          strokeWidth={strokeWidth}
          strokeLinecap="round"
          strokeLinejoin="round"
        />
      ))}
    </Svg>
  );
}

export default Icon;
```

```tsx
// src/shared/components/ui/sun-rays.tsx
import React from 'react';
import { StyleSheet, View } from 'react-native';
import Svg, { Polygon } from 'react-native-svg';

/**
 * The eight rays of the Philippine sun, fanning from just above the top edge.
 *
 * Drawn as polygons rather than a CSS conic gradient, which React Native has
 * no equivalent for. Purely decorative, so it is `pointerEvents="none"` and
 * sits behind the header's content.
 */
export function SunRays({
  color = '#FCD116',
  opacity = 0.16,
}: {
  color?: string;
  opacity?: number;
}) {
  const cx = 200;
  const cy = -40;
  const length = 460;
  const half = 4; // half-angle of each ray, in degrees

  const rays = Array.from({ length: 8 }, (_, i) => {
    const mid = (360 / 8) * i;
    const point = (deg: number) => {
      const rad = ((deg - 90) * Math.PI) / 180;
      return `${cx + Math.cos(rad) * length},${cy + Math.sin(rad) * length}`;
    };
    return `${cx},${cy} ${point(mid - half)} ${point(mid + half)}`;
  });

  return (
```

```tsx
import React from 'react';
```

```tsx
import Icon, { type IconName } from './icon';
import { H3 } from './text';
import tokens from '@/shared/theme/tokens';

type Variant = 'primary' | 'gold' | 'secondary' | 'locked';

const VARIANTS: Record<Variant, { bg: string; shadow: string; fg: string; border?: string }> = {
  primary: { bg: tokens.color.primary, shadow: tokens.color.primaryDark, fg: tokens.color.onDark },
  gold: { bg: tokens.color.gold, shadow: tokens.color.goldDark, fg: tokens.color.goldInk },
  secondary: { bg: tokens.color.surface, shadow: tokens.color.border, fg: tokens.color.inkMuted, border: tokens.color.border },

};

/**
 * `color` overrides the variant's fill — the category-coloured call to action
 * on the level and activity screens. The chunky look is a solid bottom border
 * rather than a shadow: React Native has no `box-shadow: 0 6px 0`, and this
 * renders the same on both platforms.
 */
export function Button({
  label,
  onPress,
  variant = 'primary',
  color,
  shadowColor,
  icon,
  disabled = false,
  busy = false,
  style,
}: {
  label: string;

  color?: string;
  shadowColor?: string;
  icon?: IconName;
  disabled?: boolean;
  busy?: boolean;
  style?: ViewStyle;
}) {
```
